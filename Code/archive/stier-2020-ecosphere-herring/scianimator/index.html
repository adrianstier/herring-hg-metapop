for (i in years) {
    p.data <- surveyData[which(surveyData$YEAR == i), ]
    max.z <- max(p.data$Z)
    subset <- p.data[p.data$Z > 0, ]
    print(i)
    if (length(subset[, 1]) >= 1) {
        subset <- data.frame(EID = p.data$EID, X = p.data$X, 
            Y = p.data$Y, Z = p.data$Z, loc = p.data$LOC_CODE)
        plotMap(nepacLLhigh, xlim = x.range, ylim = y.range, 
            col = "gainsboro", plt = c(0.08, 0.99, 0.08, 0.99))
        title(main = paste(i, "                         "), line = -4)
        ramp <- colorRamp(c("red", "white"))
        addBubbles(subset, type = "surface", symbol.bg = rev(heat.colors(n.colors)), 
            legend.type = "horiz", z.max = max.obs, legend.breaks = pretty(min.obs:max.obs, 
                n = n.colors), symbol.zero = FALSE, col = "grey", 
            min.size = 0.1, max.size = 0.4)
    }
    ani.pause()
}
